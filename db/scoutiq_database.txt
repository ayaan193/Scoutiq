-- ScoutIQ PostgreSQL 16 + pgvector database setup. Run with psql -v ON_ERROR_STOP=1 -f db/scoutiq_database.txt.
-- Database creation (run separately as a PostgreSQL superuser): CREATE DATABASE scoutiq;
CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pg_trgm;
CREATE EXTENSION IF NOT EXISTS unaccent;

CREATE TABLE users (user_id SERIAL PRIMARY KEY, username VARCHAR(50) NOT NULL UNIQUE CHECK (username ~ '^[a-z0-9_]{3,50}$'), password_hash VARCHAR(255) NOT NULL, role VARCHAR(10) NOT NULL CHECK (role IN ('scout','admin')), created_at TIMESTAMP NOT NULL DEFAULT now());
CREATE TABLE teams (team_id SERIAL PRIMARY KEY, team_name VARCHAR(100) NOT NULL UNIQUE, league VARCHAR(100), country VARCHAR(100));
CREATE TABLE players (player_id SERIAL PRIMARY KEY, team_id INT REFERENCES teams(team_id) ON DELETE SET NULL, name VARCHAR(100) NOT NULL, age INT CHECK (age BETWEEN 15 AND 45), position VARCHAR(30) NOT NULL, nationality VARCHAR(56), preferred_foot VARCHAR(5) CHECK (preferred_foot IN ('Left','Right','Both')), market_value_eur NUMERIC(12,0) CHECK (market_value_eur >= 0), updated_at TIMESTAMP NOT NULL DEFAULT now());
CREATE TABLE player_statistics (stat_id SERIAL PRIMARY KEY, player_id INT NOT NULL REFERENCES players(player_id) ON DELETE CASCADE, season VARCHAR(10) NOT NULL CHECK (season ~ '^[0-9]{4}-[0-9]{2}$'), goals INT NOT NULL DEFAULT 0 CHECK (goals >= 0), assists INT NOT NULL DEFAULT 0 CHECK (assists >= 0), pass_accuracy DECIMAL(5,2) CHECK (pass_accuracy BETWEEN 0 AND 100), dribbling INT CHECK (dribbling BETWEEN 0 AND 100), CONSTRAINT uq_player_season UNIQUE (player_id,season));
CREATE TABLE scouting_reports (report_id SERIAL PRIMARY KEY, player_id INT NOT NULL REFERENCES players(player_id) ON DELETE CASCADE, scout_id INT REFERENCES users(user_id) ON DELETE SET NULL, match_date DATE NOT NULL, opponent VARCHAR(100), rating SMALLINT CHECK (rating BETWEEN 1 AND 10), strengths TEXT, weaknesses TEXT, summary TEXT NOT NULL CHECK (char_length(trim(summary)) >= 30), search_tsv TSVECTOR GENERATED ALWAYS AS (setweight(to_tsvector('english',coalesce(summary,'')),'A') || setweight(to_tsvector('english',coalesce(strengths,'')),'B') || setweight(to_tsvector('english',coalesce(weaknesses,'')),'C')) STORED, embedding VECTOR(384), embedded_at TIMESTAMP, created_at TIMESTAMP NOT NULL DEFAULT now());
CREATE TABLE shortlists (shortlist_id SERIAL PRIMARY KEY, user_id INT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE, player_id INT NOT NULL REFERENCES players(player_id) ON DELETE CASCADE, priority SMALLINT NOT NULL DEFAULT 3 CHECK (priority BETWEEN 1 AND 5), notes TEXT, added_at TIMESTAMP NOT NULL DEFAULT now(), UNIQUE(user_id,player_id));
CREATE TABLE recommendations (recommendation_id SERIAL PRIMARY KEY, player_id INT NOT NULL REFERENCES players(player_id) ON DELETE CASCADE, recommended_by INT NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT, decided_by INT REFERENCES users(user_id) ON DELETE SET NULL, status VARCHAR(10) NOT NULL DEFAULT 'pending' CHECK (status IN ('pending','approved','rejected','withdrawn')), priority VARCHAR(6) NOT NULL DEFAULT 'medium' CHECK (priority IN ('low','medium','high')), justification TEXT NOT NULL CHECK (char_length(trim(justification)) >= 50), decision_note TEXT, created_at TIMESTAMP NOT NULL DEFAULT now(), decided_at TIMESTAMP, CHECK ((status='pending' AND decided_at IS NULL) OR (status<>'pending' AND decided_at IS NOT NULL AND decided_at>=created_at)));
CREATE TABLE audit_log (audit_id BIGSERIAL PRIMARY KEY, table_name VARCHAR(50) NOT NULL, record_id INT NOT NULL, action VARCHAR(6) NOT NULL CHECK (action IN ('INSERT','UPDATE','DELETE')), changed_by INT REFERENCES users(user_id) ON DELETE SET NULL, old_data JSONB, new_data JSONB, changed_at TIMESTAMP NOT NULL DEFAULT now());

CREATE UNIQUE INDEX uq_pending_recommendation ON recommendations(player_id,recommended_by) WHERE status='pending';
CREATE INDEX ix_players_filters ON players(position,age,team_id);
CREATE INDEX ix_players_name_trgm ON players USING GIN(name gin_trgm_ops);
CREATE INDEX ix_reports_search ON scouting_reports USING GIN(search_tsv);
CREATE INDEX ix_reports_embedding ON scouting_reports USING hnsw(embedding vector_cosine_ops);
CREATE INDEX ix_recommendations_queue ON recommendations(status,priority,created_at);
CREATE INDEX ix_audit_recent ON audit_log(changed_at DESC);

CREATE FUNCTION current_app_user() RETURNS INT LANGUAGE sql STABLE AS $$ SELECT nullif(current_setting('app.user_id',true),'')::INT $$;
CREATE FUNCTION current_app_role() RETURNS TEXT LANGUAGE sql STABLE AS $$ SELECT nullif(current_setting('app.role',true),'') $$;
CREATE FUNCTION trg_players_updated_at() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN NEW.updated_at=now(); RETURN NEW; END $$;
CREATE TRIGGER players_updated_at BEFORE UPDATE ON players FOR EACH ROW EXECUTE FUNCTION trg_players_updated_at();
CREATE FUNCTION trg_validate_report() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN IF NEW.match_date>CURRENT_DATE THEN RAISE EXCEPTION 'Match date cannot be in the future'; END IF; IF TG_OP='UPDATE' THEN IF current_app_role()='scout' AND OLD.created_at<now()-interval '7 days' THEN RAISE EXCEPTION 'Reports can only be edited within 7 days'; END IF; IF (NEW.summary,NEW.strengths,NEW.weaknesses) IS DISTINCT FROM (OLD.summary,OLD.strengths,OLD.weaknesses) THEN NEW.embedding=NULL; NEW.embedded_at=NULL; END IF; END IF; RETURN NEW; END $$;
CREATE TRIGGER validate_report BEFORE INSERT OR UPDATE ON scouting_reports FOR EACH ROW EXECUTE FUNCTION trg_validate_report();
CREATE FUNCTION trg_recommendation_rules() RETURNS trigger LANGUAGE plpgsql AS $$ DECLARE maker_role TEXT; decider_role TEXT; BEGIN SELECT role INTO maker_role FROM users WHERE user_id=NEW.recommended_by; IF maker_role IS DISTINCT FROM 'scout' THEN RAISE EXCEPTION 'Only scouts can recommend players'; END IF; IF NEW.decided_by IS NOT NULL THEN SELECT role INTO decider_role FROM users WHERE user_id=NEW.decided_by; IF decider_role IS DISTINCT FROM 'admin' THEN RAISE EXCEPTION 'Only admins can decide recommendations'; END IF; END IF; IF TG_OP='UPDATE' THEN IF OLD.status IN ('approved','rejected','withdrawn') THEN RAISE EXCEPTION 'A decided recommendation cannot change'; END IF; IF NEW.justification IS DISTINCT FROM OLD.justification OR NEW.recommended_by IS DISTINCT FROM OLD.recommended_by OR NEW.player_id IS DISTINCT FROM OLD.player_id THEN RAISE EXCEPTION 'Recommendation details cannot be edited'; END IF; END IF; IF NEW.status IN ('approved','rejected') AND NEW.decided_by IS NULL THEN RAISE EXCEPTION 'A decision needs an admin'; END IF; IF NEW.status='withdrawn' AND NEW.decided_by IS NOT NULL THEN RAISE EXCEPTION 'A withdrawal cannot have a deciding admin'; END IF; RETURN NEW; END $$;
CREATE TRIGGER recommendation_rules BEFORE INSERT OR UPDATE ON recommendations FOR EACH ROW EXECUTE FUNCTION trg_recommendation_rules();
CREATE FUNCTION trg_recommendation_actor() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN IF current_app_role() IS NULL THEN RETURN NEW; END IF; IF TG_OP='INSERT' THEN IF current_app_role()<>'scout' OR NEW.recommended_by<>current_app_user() THEN RAISE EXCEPTION 'Only a scout can create their own recommendation'; END IF; ELSIF NEW.status IN ('approved','rejected') THEN IF current_app_role()<>'admin' OR NEW.decided_by<>current_app_user() THEN RAISE EXCEPTION 'Only the acting admin can decide'; END IF; ELSIF NEW.status='withdrawn' THEN IF current_app_role()<>'scout' OR OLD.recommended_by<>current_app_user() THEN RAISE EXCEPTION 'Only the recommending scout can withdraw'; END IF; END IF; RETURN NEW; END $$;
CREATE TRIGGER recommendation_actor BEFORE INSERT OR UPDATE ON recommendations FOR EACH ROW EXECUTE FUNCTION trg_recommendation_actor();
CREATE FUNCTION trg_report_actor() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN IF current_app_role() IS NULL THEN RETURN CASE WHEN TG_OP='DELETE' THEN OLD ELSE NEW END; END IF; IF TG_OP='INSERT' AND (current_app_role()<>'scout' OR NEW.scout_id<>current_app_user()) THEN RAISE EXCEPTION 'Only a scout can file their own report'; END IF; IF TG_OP IN ('UPDATE','DELETE') AND current_app_role()<>'admin' AND (current_app_role()<>'scout' OR OLD.scout_id<>current_app_user()) THEN RAISE EXCEPTION 'Only the author can change this report'; END IF; RETURN CASE WHEN TG_OP='DELETE' THEN OLD ELSE NEW END; END $$;
CREATE TRIGGER report_actor BEFORE INSERT OR UPDATE OR DELETE ON scouting_reports FOR EACH ROW EXECUTE FUNCTION trg_report_actor();
CREATE FUNCTION trg_audit() RETURNS trigger LANGUAGE plpgsql AS $$ DECLARE row_id INT; BEGIN row_id=CASE TG_OP WHEN 'DELETE' THEN (to_jsonb(OLD)->>TG_ARGV[0])::INT ELSE (to_jsonb(NEW)->>TG_ARGV[0])::INT END; INSERT INTO audit_log(table_name,record_id,action,changed_by,old_data,new_data) VALUES (TG_TABLE_NAME,row_id,TG_OP,current_app_user(),CASE WHEN TG_OP='INSERT' THEN NULL ELSE to_jsonb(OLD) END,CASE WHEN TG_OP='DELETE' THEN NULL ELSE to_jsonb(NEW) END); RETURN CASE WHEN TG_OP='DELETE' THEN OLD ELSE NEW END; END $$;
CREATE TRIGGER audit_players AFTER INSERT OR UPDATE OR DELETE ON players FOR EACH ROW EXECUTE FUNCTION trg_audit('player_id');
CREATE TRIGGER audit_reports AFTER INSERT OR UPDATE OR DELETE ON scouting_reports FOR EACH ROW EXECUTE FUNCTION trg_audit('report_id');
CREATE TRIGGER audit_recommendations AFTER INSERT OR UPDATE OR DELETE ON recommendations FOR EACH ROW EXECUTE FUNCTION trg_audit('recommendation_id');

CREATE FUNCTION fn_player_score(p_goals INT,p_assists INT,p_rating NUMERIC,p_dribbling INT) RETURNS NUMERIC LANGUAGE sql IMMUTABLE AS $$ SELECT round((least(coalesce(p_goals,0),30)*1.5 + least(coalesce(p_assists,0),25)*1.2 + coalesce(p_rating,0)*4 + coalesce(p_dribbling,0)*0.2)::numeric,1) $$;
CREATE VIEW player_summary AS SELECT p.*,t.team_name,s.season,s.goals,s.assists,s.pass_accuracy,s.dribbling,r.average_rating,fn_player_score(s.goals,s.assists,r.average_rating,s.dribbling) score FROM players p LEFT JOIN teams t ON t.team_id=p.team_id LEFT JOIN LATERAL (SELECT * FROM player_statistics WHERE player_id=p.player_id ORDER BY season DESC LIMIT 1) s ON true LEFT JOIN LATERAL (SELECT round(avg(rating)::numeric,1) average_rating FROM scouting_reports WHERE player_id=p.player_id) r ON true;
CREATE VIEW scout_activity AS SELECT u.user_id,u.username,count(DISTINCT sr.report_id) reports,count(DISTINCT rec.recommendation_id) recommendations,count(DISTINCT rec.recommendation_id) FILTER (WHERE rec.status='approved') approvals FROM users u LEFT JOIN scouting_reports sr ON sr.scout_id=u.user_id LEFT JOIN recommendations rec ON rec.recommended_by=u.user_id WHERE u.role='scout' GROUP BY u.user_id,u.username;
CREATE VIEW pending_recommendations AS SELECT r.*,p.name player_name,u.username scout_name FROM recommendations r JOIN players p ON p.player_id=r.player_id JOIN users u ON u.user_id=r.recommended_by WHERE r.status='pending';
CREATE VIEW position_leaderboard AS SELECT player_id,name,position,team_name,score,RANK() OVER (PARTITION BY position ORDER BY score DESC) position_rank FROM player_summary;

CREATE FUNCTION sp_add_player_with_stats(p_team_id INT,p_name VARCHAR,p_age INT,p_position VARCHAR,p_nationality VARCHAR,p_foot VARCHAR,p_value NUMERIC,p_season VARCHAR,p_goals INT,p_assists INT,p_accuracy NUMERIC,p_dribbling INT) RETURNS INT LANGUAGE plpgsql AS $$ DECLARE new_id INT; BEGIN INSERT INTO players(team_id,name,age,position,nationality,preferred_foot,market_value_eur) VALUES(p_team_id,p_name,p_age,p_position,p_nationality,p_foot,p_value) RETURNING player_id INTO new_id; INSERT INTO player_statistics(player_id,season,goals,assists,pass_accuracy,dribbling) VALUES(new_id,p_season,p_goals,p_assists,p_accuracy,p_dribbling); RETURN new_id; END $$;
CREATE FUNCTION sp_decide_recommendation(p_id INT,p_admin INT,p_decision TEXT,p_note TEXT) RETURNS VOID LANGUAGE plpgsql AS $$ DECLARE rec recommendations%ROWTYPE; BEGIN IF p_decision NOT IN ('approved','rejected') THEN RAISE EXCEPTION 'Invalid decision'; END IF; SELECT * INTO rec FROM recommendations WHERE recommendation_id=p_id FOR UPDATE; IF NOT FOUND OR rec.status<>'pending' THEN RAISE EXCEPTION 'Recommendation is not pending'; END IF; UPDATE recommendations SET status=p_decision,decided_by=p_admin,decision_note=p_note,decided_at=now() WHERE recommendation_id=p_id; IF p_decision='approved' THEN INSERT INTO shortlists(user_id,player_id,priority,notes) VALUES(rec.recommended_by,rec.player_id,1,'Approved recommendation') ON CONFLICT (user_id,player_id) DO UPDATE SET priority=1; END IF; END $$;
CREATE FUNCTION fn_similar_players(p_player_id INT,p_limit INT DEFAULT 5) RETURNS TABLE(player_id INT,name VARCHAR,"position" VARCHAR,age INT,market_value_eur NUMERIC,distance DOUBLE PRECISION) LANGUAGE sql STABLE AS $$ WITH profile AS (SELECT avg(embedding) vector FROM scouting_reports WHERE player_id=p_player_id AND embedding IS NOT NULL) SELECT p.player_id,p.name,p.position,p.age,p.market_value_eur,min(r.embedding <=> profile.vector) distance FROM profile JOIN scouting_reports r ON r.embedding IS NOT NULL JOIN players p ON p.player_id=r.player_id WHERE p.player_id<>p_player_id AND profile.vector IS NOT NULL GROUP BY p.player_id,p.name,p.position,p.age,p.market_value_eur ORDER BY distance LIMIT p_limit $$;
